def insertion_sort(arr): # 從第二個元素開始，因為第一個元素視為已排序 for i in range(1, len(arr)): key = arr[i] # 當前要插入的元素 j = i - 1 # 將 key 插入到已排序部分的正確位置 while j >= 0 and arr[j] > key: arr[j + 1] = arr[j] # 向右移動元素 j -= 1 arr[j + 1] = key # 插入元素 return arr if __name__ == "__main__": data = [64, 34, 25, 12, 22, 11, 90] print("原始陣列:", data) sorted_data = insertion_sort(data) print("排序後陣列:", sorted_data)
